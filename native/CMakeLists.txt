cmake_minimum_required(VERSION 3.24)
project(Athanor VERSION 1.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(ATHANOR_MANUAL_MOC "Use existing generated metaobjects in a managed build" OFF)
if(ATHANOR_MANUAL_MOC)
 set(CMAKE_AUTOMOC OFF)
else()
 set(CMAKE_AUTOMOC ON)
endif()
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Widgets Quick QuickControls2 Concurrent Svg ShaderTools Network)
set(ATHANOR_CODEC_ROOT "" CACHE PATH "Native codec development files")
qt_add_executable(Athanor WIN32 main.cpp updater.cpp updater.h controller.cpp controller.h childprocess.cpp childprocess.h conversion.cpp conversion.h platform.cpp platform.h pdf_bridge.c pdf_bridge.h ui_test.cpp ui_test.h)
if(ATHANOR_MANUAL_MOC)
 target_sources(Athanor PRIVATE "${ATHANOR_CODEC_ROOT}/generated/moc_updater.cpp" "${ATHANOR_CODEC_ROOT}/generated/moc_controller.cpp" "${ATHANOR_CODEC_ROOT}/generated/moc_childprocess.cpp")
endif()
qt_add_resources(Athanor "app_qml" PREFIX "/qt/qml/Athanor" FILES Main.qml Quick.qml Preview.qml ActionButton.qml Card.qml Field.qml Spin.qml Check.qml Radio.qml SizeChoice.qml SizeAmount.qml Checker.qml Theme.qml Glyph.qml Input.qml QueuePanel.qml Hint.qml Formats.qml FormatSelect.qml FormatGuide.qml ConversionPrompts.qml qmldir)
qt_add_shaders(Athanor "reveal_shader" PREFIX "/shaders" FILES reveal.frag rounded-mask.frag checker.frag)
target_link_libraries(Athanor PRIVATE Qt6::Core Qt6::Gui Qt6::Widgets Qt6::Quick Qt6::QuickControls2 Qt6::Concurrent Qt6::Svg Qt6::Network)
target_include_directories(Athanor PRIVATE "${ATHANOR_CODEC_ROOT}/mupdf/include" "${ATHANOR_CODEC_ROOT}/avif/include")
target_include_directories(Athanor PRIVATE "${ATHANOR_CODEC_ROOT}/webp/Library/include")
target_include_directories(Athanor PRIVATE "${ATHANOR_CODEC_ROOT}/heif/include")
if(WIN32)
 target_link_libraries(Athanor PRIVATE "${ATHANOR_CODEC_ROOT}/heif.lib")
 target_compile_options(Athanor PRIVATE /utf-8)
 target_link_libraries(Athanor PRIVATE "${ATHANOR_CODEC_ROOT}/mupdf.lib" "${ATHANOR_CODEC_ROOT}/avif.lib" "${ATHANOR_CODEC_ROOT}/webp/Library/lib/libwebp.lib" "${ATHANOR_CODEC_ROOT}/webp/Library/lib/libwebpmux.lib" "${ATHANOR_CODEC_ROOT}/webp/Library/lib/libwebpdemux.lib" shell32 ole32 dwmapi)
 configure_file(athanor.rc.in "${CMAKE_CURRENT_BINARY_DIR}/athanor.rc" @ONLY)
 target_sources(Athanor PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/athanor.rc")
else()
 find_library(MUPDF_LIBRARY mupdf REQUIRED)
 find_library(AVIF_LIBRARY avif REQUIRED)
 find_library(WEBP_LIBRARY webp REQUIRED)
 find_library(WEBPMUX_LIBRARY webpmux REQUIRED)
 find_library(WEBPDEMUX_LIBRARY webpdemux REQUIRED)
 find_library(HEIF_LIBRARY heif REQUIRED)
 target_link_libraries(Athanor PRIVATE ${MUPDF_LIBRARY} ${AVIF_LIBRARY} ${WEBP_LIBRARY} ${WEBPMUX_LIBRARY} ${WEBPDEMUX_LIBRARY} ${HEIF_LIBRARY})
endif()
install(TARGETS Athanor RUNTIME DESTINATION .)

install(FILES qt.conf DESTINATION .)
